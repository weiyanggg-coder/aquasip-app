def fmt_clock(minute):
    minute = int(minute) % (24 * 60)
    return datetime(2026, 1, 1, minute // 60, minute % 60).strftime("%I:%M %p")


def _round50(x):
    return int(round(x / 50.0)) * 50


def steady_hourly(total_drank, goal, minute):
    hours_left = max(0.0, (DAY_END_MIN - minute) / 60.0)
    remaining = max(0, goal - total_drank)
    if hours_left <= 0:
        return 0, 0.0
    per_hour = int(round(remaining / hours_left / 50.0)) * 50
    return per_hour, round(hours_left, 1)


def fluid_pacing(total_drank, goal, minute):
    limit = max(goal, 1)
    day_span = DAY_END_MIN - DAY_START_MIN
    elapsed = max(0, min(day_span, minute - DAY_START_MIN))
    pace_pct = elapsed / day_span
    allowed_now = int(round(limit * pace_pct))
    used_pct = total_drank / limit
    left = limit - total_drank

    hours_left = max(0.0, (DAY_END_MIN - minute) / 60.0)
    per_hour = int(round(left / hours_left / 10.0)) * 10 if (hours_left > 0 and left > 0) else 0

    if total_drank > limit:
        status = "over"
    elif used_pct >= 0.85:
        status = "near"
    elif total_drank > allowed_now + 150:
        status = "ahead"
    else:
        status = "ok"

    return {
        "status": status,
        "used": total_drank,
        "limit": limit,
        "left": left,
        "used_pct": used_pct,
        "pace_pct": pace_pct,
        "allowed_now": allowed_now,
        "per_hour": per_hour,
    }


def pacing_message(p):
    st = p["status"]
    if st == "over":
        return f"⚠️ You are {p['used'] - p['limit']} ml OVER your fluid limit for today. Stop drinking fluids and check with your care team if you feel unwell."
    elif st == "near":
        return f"⚠️ You have used {int(p['used_pct']*100)}% of your limit. Only {p['left']} ml left for the rest of the day. Sip very slowly."
    elif st == "ahead":
        return f"⚡ You've consumed fluid faster than planned ({p['used']} ml vs {p['allowed_now']} ml expected by now). Slow down your sips."
    else:
        return f"✅ Good pacing! You have used {p['used']} ml out of {p['limit']} ml allowed."


def weight_alert(weights, today_date):
    if not weights or len(weights) < 2:
        return None
    sorted_items = sorted(weights.items())
    latest_date_str, latest_w = sorted_items[-1]

    for d_str, w in reversed(sorted_items[:-1]):
        try:
            d_obj = datetime.strptime(d_str, "%Y-%m-%d").date()
            cur_obj = datetime.strptime(latest_date_str, "%Y-%m-%d").date()
            days_diff = (cur_obj - d_obj).days
            if 1 <= days_diff <= 2:
                diff = latest_w - w
                if diff >= 1.5:
                    return ("alert", f"⚠️ Rapid weight gain detected (+{diff:.1f} kg in {days_diff} days). Contact your doctor as this may indicate fluid retention.")
                elif diff >= 1.0:
                    return ("watch", f"⚡ Weight increased by +{diff:.1f} kg over the last {days_diff} days. Monitor closely.")
            if 3 <= days_diff <= 7:
                diff = latest_w - w
                if diff >= 2.0:
                    return ("alert", f"⚠️ Weight increased by +{diff:.1f} kg over {days_diff} days. Check with your care team.")
        except Exception:
            continue
    return ("ok", "Weight stable.")


def sweat_rate(pre, post, drank, mins):
    if mins <= 0:
        return None
    weight_loss_kg = pre - post
    total_sweat_ml = (weight_loss_kg * 1000) + drank
    rate_lph = round((total_sweat_ml / 1000.0) / (mins / 60.0), 2)
    pct = round((weight_loss_kg / pre) * 100, 1)
    rehydrate_ml = int(round(max(0, weight_loss_kg * 1500) / 50.0)) * 50
    during_per_15 = int(round((total_sweat_ml / (mins / 15.0)) / 10.0)) * 10

    if pct >= 2.0:
        note = f"You lost {pct}% of body weight. You are dehydrated! Rehydrate with ~{rehydrate_ml} ml of water & electrolytes."
    else:
        note = f"Hydration lost during exercise was {pct}%, within safe bounds."

    return {
        "rate_lph": rate_lph,
        "pct": pct,
        "note": note,
        "during_per_15": during_per_15,
        "rehydrate_ml": rehydrate_ml,
    }

def predict_goal_success(by_day, minute, total_drank, goal):
    if goal <= 0:
        return None
    remaining = max(0, goal - total_drank)
    if remaining == 0:
        return {"prob": 100, "n_days": len(by_day), "typical_rest": None, "remaining": 0}
    if minute >= DAY_END_MIN or not by_day:
        return None

    bandwidth = 0.12 * goal
    w_sum, w_hit, rest_vals = 0.0, 0.0, []
    for entries in by_day.values():
        so_far = sum(a for t, a in entries if t <= minute)
        rest = sum(a for t, a in entries if t > minute)
        w = math.exp(-((so_far - total_drank) / bandwidth) ** 2)
        w_sum += w
        w_hit += w * (1.0 if rest >= remaining else 0.0)
        rest_vals.append(rest)

    hits = sum(1 for r in rest_vals if r >= remaining)
    base_rate = (hits + 0.5) / (len(rest_vals) + 1.0)
    prob = (w_hit + base_rate) / (w_sum + 1.0)
    prob = min(0.97, max(0.03, prob))
    return {
        "prob": int(round(100 * prob)),
        "n_days": len(by_day),
        "typical_rest": int(round(statistics.median(rest_vals), -1)) if rest_vals else None,
        "remaining": remaining,
    }


def predict_next_drink(by_day, minute, last_drink_minute):
    if minute >= DAY_END_MIN or not by_day:
        return None

    gaps, firsts = [], []
    for entries in by_day.values():
        times = sorted(t for t, _ in entries)
        if times:
            firsts.append(times[0])
        for a, b in zip(times, times[1:]):
            if 5 <= b - a <= 240:
                gaps.append(b - a)

    if last_drink_minute is None:
        if not firsts:
            return None
        first = int(statistics.median(firsts))
        if first > minute:
            return {"kind": "first", "minute": first, "gap": None}
        return {"kind": "overdue_first", "minute": first, "gap": None}

    if not gaps:
        return None
    gap = int(statistics.median(gaps))
    nxt = last_drink_minute + gap
    if nxt >= DAY_END_MIN:
        return {"kind": "done_for_day", "minute": None, "gap": gap}
    kind = "next" if nxt > minute else "overdue"
    return {"kind": kind, "minute": nxt, "gap": gap, "late_by": max(0, minute - nxt)}


def fetch_weather(lat, lon, timeout=4):
    try:
        r = requests.get(
            "https://api.open-meteo.com/v1/forecast",
            params={
                "latitude": lat,
                "longitude": lon,
                "current": "temperature_2m,relative_humidity_2m,apparent_temperature",
                "daily": "temperature_2m_max",
                "timezone": "auto",
                "forecast_days": 1,
            },
            timeout=timeout,
        )
        if r.status_code != 200:
            return None
        d = r.json()
        cur = d.get("current", {})
        return {
            "temp": float(cur.get("temperature_2m")),
            "feels_like": float(cur.get("apparent_temperature", cur.get("temperature_2m"))),
            "humidity": float(cur.get("relative_humidity_2m")),
            "max_temp": float(d.get("daily", {}).get("temperature_2m_max", [cur.get("temperature_2m")])[0]),
        }
    except Exception:
        return None


def adaptive_goal(base_goal, weather=None, exercise_min=0, condition="Default"):
    if condition in FLUID_RESTRICTED:
        return base_goal, [
            f"Your goal is set for {condition}. AquaSip will not raise a prescribed fluid limit "
            "for heat or exercise. Ask your doctor before changing it."
        ], False

    reasons, bonus = [], 0
    if weather:
        hot = weather.get("max_temp", weather.get("temp", 0))
        heat = min(500, max(0, (hot - 26) * 50))
        heat = _round50(heat)
        if heat > 0:
            bonus += heat
            reasons.append(f"+{heat} ml for heat (today's high is about {hot:.0f} C)")
        if weather.get("humidity", 0) >= 80 and weather.get("temp", 0) >= 26:
            bonus += 100
            reasons.append(f"+100 ml for humidity ({weather['humidity']:.0f}%)")
    if exercise_min and exercise_min > 0:
        ex = min(1500, _round50(exercise_min / 30.0 * 350))
        if ex > 0:
            bonus += ex
            reasons.append(f"+{ex} ml for {int(exercise_min)} min of exercise")
    if not reasons:
        reasons.append("No adjustment needed today.")
    return base_goal + bonus, reasons, True


def fluid_limit_status(total_drank, goal, condition):
    if condition not in FLUID_RESTRICTED or goal <= 0:
        return None
    pct = total_drank / goal
    if pct > 1.0:
        return ("over", f"You are {total_drank - goal} ml over your {condition} fluid limit of {goal} ml. "
                        "Avoid more fluids unless your doctor said otherwise, and contact your care team "
                        "if you feel breathless, swollen or unwell.")
    if pct >= 0.85:
        return ("near", f"You have about {goal - total_drank} ml left of your {goal} ml limit. "
                        "Try small sips or ice chips to spread it out for the rest of the day.")
    return ("ok", f"{goal - total_drank} ml left of your {goal} ml daily limit.")

def build_smart_schedule(by_day, goal, cup, condition="Default", max_alarms=8):
    cup = max(int(cup or 200), 50)
    n = max(4, min(max_alarms, round(goal / cup)))
    if condition in FLUID_RESTRICTED:
        n = min(n, 5)

    slots = list(range(7 * 60 + 30, 21 * 60 + 31, 30))
    n_days = max(len(by_day), 1)
    fair = goal * 90 / (DAY_END_MIN - DAY_START_MIN)

    scored = []
    for s in slots:
        typical = sum(a for ents in by_day.values() for t, a in ents if s - 45 <= t < s + 45) / n_days
        score = max(0.0, fair - typical) + 0.15 * fair
        scored.append((score, s, typical))

    min_gap = max(60, int(0.7 * (21 * 60 - 7 * 60) / n))
    picked = [(slots[0], None)]
    for score, s, typical in sorted(scored, reverse=True):
        if len(picked) >= n:
            break
        if all(abs(s - p) >= min_gap for p, _ in picked):
            picked.append((s, typical))

    per = max(50, _round50(goal / len(picked)))
    out = []
    for s, typical in sorted(picked):
        if typical is None:
            why = "Start the day hydrated"
        elif typical < 0.5 * fair:
            why = f"You usually drink only ~{int(round(typical, -1))} ml around now"
        else:
            why = "Keeps your intake evenly spread"
        out.append({"time": fmt_clock(s), "minute": s, "amount": per, "why": why})
    return out


def weekly_report(by_day, goal, today):
    totals = {d: sum(a for _, a in ents) for d, ents in by_day.items()}
    if len(totals) < 3:
        return None

    dates = sorted(totals)
    vals = [totals[d] for d in dates]
    t_str = today.strftime("%Y-%m-%d")
    wk_cut = (today - timedelta(days=7)).strftime("%Y-%m-%d")
    pw_cut = (today - timedelta(days=14)).strftime("%Y-%m-%d")
    this_wk = [totals[d] for d in dates if wk_cut <= d < t_str]
    prev_wk = [totals[d] for d in dates if pw_cut <= d < wk_cut]

    x0 = datetime.strptime(dates[0], "%Y-%m-%d")
    xs = [(datetime.strptime(d, "%Y-%m-%d") - x0).days for d in dates]
    mx, my = statistics.mean(xs), statistics.mean(vals)
    den = sum((x - mx) ** 2 for x in xs)
    slope = sum((x - mx) * (y - my) for x, y in zip(xs, vals)) / den if den else 0.0

    sd = statistics.pstdev(vals)
    cv = sd / my if my else 0
    consistency = "very consistent" if cv < 0.15 else ("fairly consistent" if cv < 0.3 else "up and down")

    by_wd = {}
    for d in dates:
        by_wd.setdefault(datetime.strptime(d, "%Y-%m-%d").strftime("%A"), []).append(totals[d])
    wd_avg = {k: statistics.mean(v) for k, v in by_wd.items()}
    best_wd = max(wd_avg, key=wd_avg.get) if len(wd_avg) >= 3 else None
    worst_wd = min(wd_avg, key=wd_avg.get) if len(wd_avg) >= 3 else None

    unusual = []
    if len(vals) >= 5 and sd > 0:
        for d in dates:
            z = (totals[d] - my) / sd
            if z <= -1.5:
                unusual.append((d, totals[d]))

    hit_rate = sum(1 for v in vals if v >= goal) / len(vals)

    lines = []
    if this_wk and prev_wk:
        a, b = statistics.mean(this_wk), statistics.mean(prev_wk)
        ch = (a - b) / b * 100 if b else 0
        word = "up" if ch > 3 else ("down" if ch < -3 else "about the same as")
        if word == "about the same as":
            lines.append(f"This week you averaged {a:,.0f} ml/day, about the same as last week.")
        else:
            lines.append(f"This week you averaged {a:,.0f} ml/day, {word} {abs(ch):.0f}% from last week.")
    else:
        lines.append(f"You averaged {my:,.0f} ml/day over {len(vals)} logged days.")

    if abs(slope) >= 15:
        lines.append(f"Trend: {'rising' if slope > 0 else 'falling'} by about {abs(slope):.0f} ml per day.")
    else:
        lines.append("Trend: steady, no big rise or drop.")
    lines.append(f"Your intake is {consistency} from day to day, and you hit your goal on {hit_rate * 100:.0f}% of logged days.")
    if best_wd and worst_wd and best_wd != worst_wd:
        lines.append(f"Best day: {best_wd} (~{wd_avg[best_wd]:,.0f} ml). Weakest day: {worst_wd} (~{wd_avg[worst_wd]:,.0f} ml).")
    if unusual:
        u = ", ".join(datetime.strptime(d, "%Y-%m-%d").strftime("%a %d %b") for d, _ in unusual[:3])
        lines.append(f"Unusually low days detected: {u}. Were you busy, travelling or unwell?")

    tip = None
    if worst_wd:
        tip = f"Set an extra reminder on {worst_wd}s, your weakest day."
    if hit_rate < 0.5:
        tip = "You're reaching your goal less than half the time. Try the AI Smart Schedule in Settings."

    return {
        "lines": lines,
        "tip": tip,
        "avg": int(my),
        "slope": round(slope, 1),
        "cv": round(cv, 2),
        "hit_rate": int(round(hit_rate * 100)),
        "days": len(vals),
        "series": [(d, totals[d]) for d in dates],
    }

def detect_lang(text):
    words = [w.strip("?!.,") for w in text.lower().split()]
    hits = sum(1 for w in words if w in _MS_WORDS)
    return "ms" if hits >= 2 or (hits == 1 and len(words) <= 3) else "en"

def instant_reply(question, ctx):
    """
    Ultra-fast pattern matching for introductions ('i am wei yang'), direct sip checks,
    greetings, gratitude, feelings, and targeted hydration queries.
    """
    q = (question or "").strip().lower()
    total = ctx.get("total", 0)
    goal = ctx.get("goal", 2000)
    cond = ctx.get("condition", "Default")
    cup = ctx.get("cup", 200)
    remaining = max(0, goal - total)
    suggested = ctx.get("suggested") or min(remaining, 300)

    # 1. Name Matcher ("i am wei yang", "my name is...", "call me...")
    name_match = re.search(r"^(?:i am|i'm|my name is|call me|nama saya)\s+([a-zA-Z\s]{2,30})$", q, re.IGNORECASE)
    if name_match:
        name = name_match.group(1).strip().title()
        if total == 0:
            return f"Nice to meet you, {name}! 👋 I'm your AquaSip AI companion. You haven't logged any water yet today out of your {goal} ml target. Ready to start with a glass?"
        else:
            return f"Nice to meet you, {name}! 👋 You've logged {total} ml of your {goal} ml goal so far today ({remaining} ml remaining). How can I help you today?"

    # 2. Action Query: "should i drink water now", "can i drink now"
    if re.search(r"\b(should|can|shall|is it time to)\s+(i|we)\s+drink\b", q) or re.search(r"\bshould i drink\b", q) or "drink water now" in q:
        if cond in FLUID_RESTRICTED and remaining <= 0:
            return f"You have reached your prescribed fluid limit of {goal} ml for today ({cond}). Avoid additional drinks unless advised by your doctor."
        if remaining <= 0:
            return f"You've already hit your {goal} ml goal today! Excellent job 🎉 If you feel genuinely thirsty, take a small sip, but you're all set."
        if total == 0:
            return f"Yes, absolutely! You haven't drunk any water yet today. Having a cup ({cup} ml) right now is a great way to start."
        if suggested and suggested > 0:
            return f"Yes! You have {remaining} ml remaining today. Aim for about {suggested} ml over the next 2 hours to stay on track."
        return f"Yes! You have {remaining} ml left to reach your {goal} ml target today. Take a sip!"

    # 3. Target Query: "how much water to drink now"
    if "how much water to drink now" in q or "how much water should i drink" in q:
        if remaining <= 0:
            return f"You've completed your {goal} ml target for today! No extra water needed."
        return f"You have {remaining} ml left today. I suggest drinking about {suggested} ml in the next 2 hours (~{round(suggested/cup, 1)} cups)."

    # 4. Small Talk & Greetings
    if q in ("hi", "hello", "hey", "hai", "helo", "good morning", "good afternoon", "good evening", "salam"):
        return f"Hello there! 👋 I'm your AquaSip AI companion. Ask me anything about daily life, general advice, or your {goal} ml hydration target (you're at {total} ml today)."

    if q in ("how are you", "how are you doing", "apa khabar", "how's it going"):
        return f"I'm doing great, thank you for asking! 😊 Ready to help you with study/work tips, health questions, or hitting your {goal} ml hydration goal!"

    if q in ("thank you", "thanks", "terima kasih", "tq", "thx", "appreciate it"):
        return "You're very welcome! Let me know if you need any other advice or tips. Stay refreshed! 💧"

    return None


def coach_reply(question, ctx):
    """
    Offline coach reply engine. Checks:
    1. Instant matcher (introductions, direct action sips, greetings)
    2. Medical safety symptoms (dizziness, chest pain, swelling)
    3. Exhaustive FAQ knowledge base (100+ weird & standard topics)
    4. General daily life AI fallback (study/focus, sleep, food, weather, fitness)
    """
    lang = detect_lang(question)
    total, goal, cond = ctx.get("total", 0), ctx.get("goal", 2000), ctx.get("condition", "Default")
    restricted = cond in FLUID_RESTRICTED
    remaining = max(0, goal - total)
    q = (question or "").strip().lower()

    # 1. Check Instant Reply Matcher
    inst = instant_reply(question, ctx)
    if inst:
        return inst

    # 2. Symptoms & Emergency Warning Check
    if any(k in q for k in ["dizzy", "pening", "headache", "sakit kepala", "faint", "pengsan", "chest", "dada", "breath", "sesak", "swell", "bengkak", "vomit", "muntah"]):
        msg = _L(lang,
                 "I can't diagnose medical symptoms. If you feel faint, have chest pain or severe shortness of breath, get emergency medical help (999). ",
                 "Saya tak boleh diagnos simptom perubatan. Kalau rasa nak pitam, sakit dada atau sesak nafas, dapatkan bantuan kecemasan (999). ")
        if restricted:
            msg += _L(lang, f"With {cond}, sudden swelling or breathlessness can mean fluid retention. Contact your doctor immediately.",
                      f"Untuk {cond}, bengkak tiba-tiba atau sesak nafas boleh jadi tanda cecair berlebihan, hubungi doktor segera.")
        else:
            msg += _L(lang, "Mild headaches or dizziness can stem from dehydration. Drink a glass of water and rest, but consult a doctor if it persists.",
                      "Sakit kepala atau pening ringan boleh berpunca daripada kurang air: minum air dan rehat, tapi jumpa doktor kalau berlarutan.")
        return msg

    # 3. Exhaustive FAQ Search
    for keys, en, ms in _EXHAUSTIVE_FAQ:
        if any(re.search(r"(?<![a-z0-9])" + re.escape(k) + r"(?![a-z0-9])", q) for k in keys):
            ans = _L(lang, en, ms)
            if restricted:
                ans += " " + _L(lang, f"Note: With {cond}, every liquid counts toward your strict {goal} ml limit.",
                                f"Nota: Untuk {cond}, semua cecair dikira dalam had {goal} ml anda.")
            return ans

    # 4. General Daily Knowledge & Lifestyle Responses
    if any(k in q for k in ["study", "focus", "work", "productivity", "concentrate", "exam", "belajar"]):
        return _L(
            lang,
            f"To boost focus during study or work: try the Pomodoro method (25m work, 5m break), eliminate phone distractions, and keep water nearby. Dehydration reduces focus by 15-20%! You are at {total}/{goal} ml today.",
            f"Untuk tingkatkan fokus masa belajar/kerja: guna teknik Pomodoro (25m fokus, 5m rehat) dan simpan air dekat meja. Kurang air mengurangkan fokus! Anda dah minum {total}/{goal} ml hari ini.",
        )

    if any(k in q for k in ["tired", "sleep", "fatigue", "sleepy", "antuk", "penat", "letih"]):
        return _L(
            lang,
            f"Feeling tired often stems from poor sleep, stress, or mild dehydration. Try getting fresh air, sleeping 7-8 hours, and having a cool glass of water. You have {remaining} ml left for your goal.",
            f"Rasa penat/mengantuk boleh disebabkan kurang tidur atau kurang air. Cuba ambil udara segar dan minum segelas air sejuk. Baki goal anda ialah {remaining} ml.",
        )

    if any(k in q for k in ["food", "eat", "makan", "diet", "meal", "hungry", "lapar"]):
        return _L(
            lang,
            f"A balanced diet with lean protein, veggies, complex carbs, and hydrating fruits (watermelon, oranges, cucumbers) keeps energy steady. Sip water between meals for good digestion!",
            f"Pemakanan seimbang dengan protein, sayur, dan buah tinggi air (tembikai, oren, timun) mengekalkan tenaga. Minum air antara waktu makan untuk penghadaman!",
        )

    if any(k in q for k in ["weather", "hot", "panas", "rain", "hujan", "cuaca", "sun"]):
        return _L(
            lang,
            f"Hot and humid weather increases sweat fluid loss. Stay in cool areas, wear light clothing, and sip water regularly. You have {remaining} ml remaining toward your {goal} ml target today.",
            f"Cuaca panas menyebabkan badan berpeluh lebih banyak. Duduk di tempat teduh dan minum air secara berperingkat. Baki goal anda ialah {remaining} ml.",
        )

    if any(k in q for k in ["exercise", "gym", "workout", "run", "lari", "senam", "sport", "sukan"]):
        return _L(
            lang,
            f"For workouts: drink ~400-600 ml 2 hours before exercise, sip 150-250 ml every 15-20 min during activity, and rehydrate after. Check the Adaptive Goal card to adjust for exercise!",
            f"Untuk senaman: minum 400-600 ml sebelum senaman, hirup 150-250 ml setiap 15-20 minit masa senaman, dan minum semula selepasnya.",
        )

    if any(k in q for k in ["who are you", "what can you do", "help", "tolong", "fungsi"]):
        return _L(
            lang,
            f"I am your AquaSip AI companion! You can ask me daily life questions, productivity tips, diet/health advice, weird hydration facts, or check your drinking progress anytime.",
            f"Saya companion AI AquaSip anda! Anda boleh tanya soalan harian, petua fokus/kerja, nasihat pemakanan, fakta hidrasi pelik, atau semak progres hidrasi bila-bila masa.",
        )

    # 5. Smart General Conversational Fallback
    pct = int(total / goal * 100) if goal else 0
    return _L(
        lang,
        f"That's an interesting topic! I'm here to help with daily advice, lifestyle tips, productivity, general knowledge, or your hydration progress. You're currently at {total} ml of your {goal} ml target ({pct}% complete, {remaining} ml left). What else would you like to know?",
        f"Soalan yang menarik! Saya sedia bantu dengan petua harian, gaya hidup sihat, atau rutin hidrasi anda. Hari ini anda dah minum {total} ml daripada goal {goal} ml ({pct}%, baki {remaining} ml). Ada soalan lain?",
    )

_GEMINI_FALLBACK_MODELS = ["gemini-flash-latest", "gemini-2.5-flash", "gemini-2.0-flash"]


def llm_coach_reply(question, ctx, history, api_key, model="gemini-flash-latest", timeout=15):
    if not api_key:
        return None, "no key"
    cond = ctx.get("condition", "Default")
    facts = (
        f"Today's intake: {ctx.get('total')} ml. Goal: {ctx.get('goal')} ml. Cup size: {ctx.get('cup')} ml. "
        f"Health setting: {cond}. Status: {ctx.get('status_msg', '')}."
    )
    system = (
        "You are AquaSip, a careful, high-quality hydration assistant. Answer the actual question directly before adding context; never answer a specific question with generic small talk. "
        "Recognize paraphrases, incomplete wording, spelling errors, English, Bahasa Melayu, and code-switching. "
        "For age questions, ask for age if missing; when age is known, give a relevant population reference and explicitly distinguish TOTAL water (food plus all drinks) from plain drinking water. "
        "For during-exercise questions, explain that needs depend on activity, duration, heat, humidity, and sweat rate; give practical safe guidance without pretending one exact volume fits everyone. "
        "For before/after exercise, pregnancy, infants, older adults, illness, and hot weather, answer the specific topic. State uncertainty when information is missing. "
        "Never invent a source, diagnose, promise a cure, tell a user to force water, or override clinician-prescribed fluid limits. Warn against overdrinking when relevant. "
        "Use concise but complete answers (normally 3-6 sentences), headings or bullets when useful, and the same language as the user. "
        f"USER HYDRATION CONTEXT: {facts}"
    )
    contents = []
    for role, text in history[-6:]:
        contents.append({"role": "user" if role == "user" else "model", "parts": [{"text": text}]})
    contents.append({"role": "user", "parts": [{"text": question}]})
    body = {"system_instruction": {"parts": [{"text": system}]}, "contents": contents,
            "generationConfig": {"temperature": 0.6, "maxOutputTokens": 450}}

    last_err = None
    for m in [model] + [x for x in _GEMINI_FALLBACK_MODELS if x != model]:
        try:
            r = requests.post(
                f"https://generativelanguage.googleapis.com/v1beta/models/{m}:generateContent",
                params={"key": api_key}, json=body, timeout=timeout,
            )
            if r.status_code == 404:
                last_err = f"model {m} not found"
                continue
            if r.status_code != 200:
                return None, f"HTTP {r.status_code}"
            parts = r.json()["candidates"][0]["content"]["parts"]
            text = "".join(p.get("text", "") for p in parts).strip()
            if text:
                return text, None
            last_err = "empty reply"
        except requests.exceptions.RequestException:
            return None, "no internet connection"
        except Exception as e:
            return None, str(e)[:80]
    return None, last_err

def _aq_normalize(text):
    """Normalize Unicode, punctuation, whitespace, and common informal forms."""
    value = _aq_unicodedata.normalize("NFKC", str(text or "")).lower()
    value = value.replace("’", "'").replace("–", "-").replace("—", "-")
    value = re.sub(r"\b(i'm|im)\b", "i am", value)
    value = re.sub(r"\b(whats|what's)\b", "what is", value)
    value = re.sub(r"\b(dont|don't)\b", "do not", value)
    value = re.sub(r"\b(wanna)\b", "want to", value)
    value = re.sub(r"\b(gonna)\b", "going to", value)
    value = re.sub(r"[^\w\s.%+-]", " ", value, flags=re.UNICODE)
    value = re.sub(r"\s+", " ", value).strip()
    return value


def _aq_has_phrase(question, phrase):
    """Match a phrase without requiring the entire question to be identical."""
    q = _aq_normalize(question)
    p = _aq_normalize(phrase)
    if not p:
        return False
    if p in q:
        return True
    # Permit punctuation/spacing variants, not arbitrary fuzzy matches for safety.
    compact_q = re.sub(r"\s+", "", q)
    compact_p = re.sub(r"\s+", "", p)
    return compact_p in compact_q


def _aq_any(question, phrases):
    return any(_aq_has_phrase(question, phrase) for phrase in phrases)


def _aq_language(question):
    """Return Malay when Malay markers dominate; otherwise default to English."""
    q = _aq_normalize(question)
    malay = [
        "berapa", "air", "minum", "patut", "perlu", "sehari", "umur", "semasa", "ketika",
        "bersenam", "saya", "nak", "boleh", "macam", "mana", "kenapa", "untuk", "dengan",
        "tak", "tidak", "lebih", "kurang", "berapa banyak", "cawan", "botol", "panas", "sakit",
    ]
    hits = sum(1 for term in malay if _aq_has_phrase(q, term))
    return "ms" if hits >= 2 else "en"


def _aq_extract_age(question):
    """Extract explicit ages such as 15, 15 years old, umur 15, or age: 15."""
    q = _aq_normalize(question)
    patterns = [
        r"\b(?:i am|im|age|aged|umur|berumur|for a|for an|at age|age is)\s*(\d{1,3})\b",
        r"\b(\d{1,3})[- ]*(?:years?[- ]+old|year[- ]+old|yrs?[- ]+old|yo|y/o|tahun)\b",
        r"\b(?:child|kid|teen|teenager|boy|girl|anak|budak|remaja)\s*(?:aged|age|umur)?\s*(\d{1,3})\b",
    ]
    for pattern in patterns:
        match = re.search(pattern, q)
        if match:
            age = int(match.group(1))
            if 0 <= age <= 120:
                return age
    # A bare age is only accepted when the question clearly asks for age-based advice.
    if _aq_any(q, ["at what age", "what age", "age", "umur", "years old", "year old"]):
        nums = re.findall(r"\b(\d{1,3})\b", q)
        for raw in nums:
            age = int(raw)
            if 0 <= age <= 120:
                return age
    return None


def _aq_extract_sex(question, ctx=None):
    """Use explicit question wording first; profile context is only a fallback."""
    q = _aq_normalize(question)
    if _aq_any(q, ["female", "girl", "girls", "woman", "women", "she", "her", "perempuan", "wanita"]):
        return "female"
    if _aq_any(q, ["male", "boy", "boys", "man", "men", "he", "him", "lelaki", "laki-laki"]):
        return "male"
    if isinstance(ctx, dict):
        value = str(ctx.get("sex", ctx.get("gender", ""))).lower()
        if value in {"female", "f", "girl", "woman"}:
            return "female"
        if value in {"male", "m", "boy", "man"}:
            return "male"
    return None


def _aq_age_reference(age, sex=None):
    """Select the closest relevant reference entry for a supplied age."""
    if age is None:
        return None
    if age < 1:
        return AQ_AGE_WATER_REFERENCE["infant_0_6"]
    if age <= 3:
        return AQ_AGE_WATER_REFERENCE["child_1_3"]
    if age <= 8:
        return AQ_AGE_WATER_REFERENCE["child_4_8"]
    if age <= 13:
        key = "child_9_13_female" if sex == "female" else "child_9_13_male"
        return AQ_AGE_WATER_REFERENCE[key]
    if age <= 18:
        key = "teen_14_18_female" if sex == "female" else "teen_14_18_male"
        return AQ_AGE_WATER_REFERENCE[key]
    if age <= 50:
        key = "adult_female" if sex == "female" else "adult_male"
        return AQ_AGE_WATER_REFERENCE[key]
    key = "older_female" if sex == "female" else "older_male"
    return AQ_AGE_WATER_REFERENCE[key]


def _aq_age_answer(question, ctx):
    """Answer age-specific hydration questions with clear units and caveats."""
    age = _aq_extract_age(question)
    sex = _aq_extract_sex(question, ctx)
    q = _aq_normalize(question)
    if age is None and not (_aq_any(q, AQ_INTENT_ALIASES["age_amount"])
                            or _aq_any(q, ["at what age", "what age", "kids drink more water", "water for baby", "water for babies", "baby water intake", "infant water intake"])):
        return None
    if age is None and _aq_any(q, ["baby", "babies", "infant", "newborn", "bayi"]):
        return ("Babies have special hydration needs. Breast milk or infant formula is the main fluid during the first year, and extra plain water can be unsafe at some ages. Ask a paediatric clinician about the baby's exact age and feeding.")
    if age is None:
        return (
            "Daily water needs depend on age, sex, activity, food, and heat. Tell me the person's age "
            "(and whether the reference should be for a boy/man or girl/woman), and I can give a more relevant estimate. "
            "Reference values usually count water from both food and drinks—not only plain water."
        )
    if age < 1:
        return ("For babies under 12 months, needs depend on exact age, feeding, and health. Breast milk or infant formula is the main fluid; ask a paediatric clinician before giving extra plain water.")
    ref = _aq_age_reference(age, sex)
    sex_note = ""
    if 9 <= age <= 18 and sex is None:
        ref_m = _aq_age_reference(age, "male")
        ref_f = _aq_age_reference(age, "female")
        answer = (
            f"For age {age}, broad population references are about {ref_f['total_l']:.1f} L/day "
            f"of TOTAL water for girls and {ref_m['total_l']:.1f} L/day for boys. "
            "TOTAL water includes water in food and all beverages, not just plain water. "
            "These are population reference values, not compulsory targets."
        )
    else:
        total_ml = int(ref["total_l"] * 1000)
        bev_ml = int(ref["beverages_l"] * 1000)
        answer = (
            f"For age {age}, a useful population reference is about {total_ml:,} mL ({ref['total_l']:.1f} L) of TOTAL water per day. "
            f"That includes water in food; the reference estimate from all beverages is about {bev_ml:,} mL ({ref['beverages_l']:.1f} L), "
            f"which includes drinks other than plain water. {ref['note']}"
        )
    if age >= 9:
        answer += " In hot, humid weather or during exercise, needs can rise; drink regularly and respond to thirst rather than forcing a fixed amount."
    if isinstance(ctx, dict) and ctx.get("condition") in FLUID_RESTRICTED:
        answer += f" Because the profile lists {ctx.get('condition')}, follow the clinician's prescribed fluid limit instead of using a general reference."
    answer += f" Source: National Academies water reference ({AQ_SOURCE_LINKS['national_academies_water']})."
    return answer


def _aq_exercise_answer(question, ctx):
    """Contextual hydration guidance for before, during, and after activity."""
    q = _aq_normalize(question)
    if not (_aq_any(q, AQ_INTENT_ALIASES["exercise_during"])
            or _aq_any(q, AQ_INTENT_ALIASES["exercise_before"])
            or _aq_any(q, AQ_INTENT_ALIASES["exercise_after"])
            or (_aq_any(q, AQ_ACTIVITY_WORDS) and _aq_any(q, AQ_WATER_WORDS))):
        return None
    if isinstance(ctx, dict) and ctx.get("condition") in FLUID_RESTRICTED:
        return (f"Your profile lists {ctx.get('condition')}, so general exercise-fluid targets may not be safe for you. "
                "Follow your clinician's fluid plan and ask them how to adapt it for exercise or heat.")
    before = _aq_any(q, AQ_INTENT_ALIASES["exercise_before"])
    after = _aq_any(q, AQ_INTENT_ALIASES["exercise_after"])
    during = _aq_any(q, AQ_INTENT_ALIASES["exercise_during"])
    if before and not after and not during:
        return ("Before exercise: start normally hydrated and drink water with your meal or in the hours beforehand; "
                "avoid chugging a large volume right before starting. The right amount depends on heat, duration, body size, and sweat rate. "
                "For a short, easy session, water is usually enough.")
    if after and not before and not during:
        return ("After exercise: drink gradually with food and over the next few hours. If you have a reliable pre/post exercise weight, "
                "the change can help estimate sweat loss, but don't force fluids or aim to gain weight during exercise. "
                "After long, hot, or very sweaty sessions, a normal meal or an appropriate electrolyte drink may help replace sodium.")

def _aq_goal_now_answer(question, ctx):
    q = _aq_normalize(question)
    if not _aq_any(q, AQ_INTENT_ALIASES["goal_now"]):
        return None
    total = max(0, int((ctx or {}).get("total", 0) or 0))
    goal = max(1, int((ctx or {}).get("goal", 2000) or 2000))
    cup = max(1, int((ctx or {}).get("cup", 200) or 200))
    remaining = max(0, goal - total)
    if isinstance(ctx, dict) and ctx.get("condition") in FLUID_RESTRICTED:
        return f"You've logged {total:,} mL against the {goal:,} mL limit in your profile ({remaining:,} mL remaining). Follow your clinician's instructions; don't exceed the limit to satisfy reminders."
    if remaining == 0:
        return f"Your app goal of {goal:,} mL is logged for today. You don't need to force extra water to hit a number; if you're thirsty, drink normally unless a clinician has told you to restrict fluids."
    suggested = min(remaining, max(100, min(cup, 300)))
    return (f"You've logged {total:,} mL of your app goal of {goal:,} mL, leaving {remaining:,} mL. If you're thirsty, a normal glass (around {suggested:,} mL) is a reasonable next drink. "
            "Spread drinks out rather than chugging to catch up. Your app goal is a tracking target, not a medical prescription.")

def _aq_haze_answer(question, ctx):
    """Handle haze/air-pollution hydration questions without inventing extra water needs."""
    q = _aq_normalize_spelling(question)
    haze_terms = AQ_INTENT_ALIASES.get("haze", [])
    if not _aq_any(q, haze_terms):
        return None

    # A known age should add a relevant daily reference, but haze itself does not
    # create a universal extra-water requirement. Keep the two concepts distinct.
    age = _aq_extract_age(question)
    sex = _aq_extract_sex(question, ctx)
    context = ctx if isinstance(ctx, dict) else {}
    if context.get("condition") in FLUID_RESTRICTED:
        return (

        )

    age_sentence = ""
    if age is not None and age >= 1:
        if 9 <= age <= 18 and sex is None:
            female = _aq_age_reference(age, "female")
            male = _aq_age_reference(age, "male")

        age_sentence = (
            " For an infant, do not add plain water without paediatric advice; feeding and exact "

        )

    return (

        "avoid forcing water or chugging large amounts."

def _aq_route_question(question, ctx):
    """Run specific and safety-sensitive intents before generic keyword matches."""
    q = _aq_normalize(question)
    if not q:
        return "Ask me a hydration question—for example, 'How much water should a 15-year-old drink?' or 'How much should I drink during football?'"

    # Priority 1: emergency / fluid-restriction safety.
    result = _aq_safety_answer(q, ctx)
    if result:
        return result

    # Priority 2: haze is an air-quality topic, not an exercise or age-only query.
    # It must be routed before age matching so an age modifier does not erase haze context.
    result = _aq_haze_answer(q, ctx)
    if result:
        return result

    # Priority 3: explicit exercise timing, including phrasing with no question mark.
    result = _aq_during_keyword_answer(q, ctx)
    if result:
        return result
    if (_aq_any(q, AQ_INTENT_ALIASES["exercise_before"])
            or _aq_any(q, AQ_INTENT_ALIASES["exercise_after"])):
        result = _aq_exercise_answer(q, ctx)
        if result:
            return result

    # Priority 3: age-specific recommendations must beat generic 'water' matches.
    if _aq_extract_age(q) is not None and _aq_any(q, AQ_WATER_WORDS):
        result = _aq_age_answer(q, ctx)
        if result:
            return result

    # Priority 4: explicit daily amount / age amount.
    result = _aq_daily_intake_answer(q, ctx)
    if result:
        return result

    # Priority 5: targeted hydration and lifestyle intents.
    for handler in (
        _aq_dehydration_answer,
        _aq_schedule_answer,
        _aq_goal_now_answer,
        _aq_water_quality_answer,
        _aq_beverage_answer,
        _aq_hot_weather_answer,
        _aq_illness_answer,
        _aq_bottle_answer,
    ):
        result = handler(q, ctx)
        if result:
            return result

    # Broaden the old FAQ keyword matching without discarding the original KB.
    for keys, en, ms in _EXHAUSTIVE_FAQ:
        if _aq_any(q, keys):
            answer = _L(_aq_language(q), en, ms)
            if isinstance(ctx, dict) and ctx.get("condition") in FLUID_RESTRICTED:
                answer += f" Note: follow your prescribed fluid limit for {ctx.get('condition')}."
            return answer

    return None

def _aq_extended_faq_answer(question, ctx):
    """Rank extended FAQ matches by phrase specificity to reduce false matches."""
    q = _aq_normalize(question)
    matches = []
    for topic, aliases, answer in AQ_EXTENDED_FAQ:
        for alias in aliases:
            if _aq_has_phrase(q, alias):
                matches.append((len(_aq_normalize(alias)), topic, answer))
    if not matches:
        return None
    matches.sort(reverse=True)
    answer = matches[0][2]
    if isinstance(ctx, dict) and ctx.get("condition") in FLUID_RESTRICTED:
        answer += f" Since your profile lists {ctx.get('condition')}, follow your clinician's fluid plan over general guidance."
    if _aq_language(question) == "ms":
        answer = _aq_translate_extended_to_malay(answer, question)
    return answer

    """Expose a simple intent label for diagnostics, logging, and competition demos."""

        ("age_based_intake", lambda: _aq_extract_age(q) is not None and _aq_any(q, AQ_WATER_WORDS)),
        ("daily_water_amount", lambda: _aq_any(q, AQ_INTENT_ALIASES["age_amount"])),

        ("bottle_hygiene", lambda: _aq_any(q, AQ_INTENT_ALIASES["bottle"])),
    ]
    for label, predicate in priority:
        try:
            if predicate():
                return label
        except Exception:
            continue
    q = _aq_normalize(question)
    if _aq_any(q, ["at what age", "what age", "kids drink more water", "water for baby", "water for babies", "baby water intake", "infant water intake"]):
        return "daily_water_amount"

    return "general_or_unknown"

def _aq_normalize_spelling(text):
    value = _aq_normalize(text)
    for wrong, right in AQ_SPELLING_NORMALIZATIONS.items():
        value = re.sub(r"(?<!\w)" + re.escape(wrong) + r"(?!\w)", right, value)
    return value


def _aq_fuzzy_alias_match(question, aliases, threshold=0.91):
    """Conservative typo-tolerant matching for short known phrases only."""
    q = _aq_normalize_spelling(question)
    if _aq_any(q, aliases):
        return True
    # Avoid fuzzy matching single words like 'age' or 'water' in isolation.
    tokens = q.split()
    if len(tokens) < 3:
        return False
    for alias in aliases:
        candidate = _aq_normalize_spelling(alias)
        if len(candidate) < 12:
            continue
        if _AQSequenceMatcher(None, q, candidate).ratio() >= threshold:
            return True
    return False

def aq_validate_context(ctx):
    """Normalize context values without mutating the caller's dictionary."""
    raw = ctx if isinstance(ctx, dict) else {}
    normalized = dict(raw)
    for key, default in (("total", 0), ("goal", 2000), ("cup", 200)):
        try:
            normalized[key] = int(float(raw.get(key, default) or default))
        except (TypeError, ValueError, OverflowError):
            normalized[key] = default
    normalized["total"] = max(0, normalized["total"])
    normalized["goal"] = max(1, normalized["goal"])
    normalized["cup"] = max(1, normalized["cup"])
    return normalized


def aq_response_quality_check(answer):
    """Return diagnostic warnings, not a claim that a medical answer is validated."""
    warnings = []
    text = str(answer or "")
    if not text.strip():
        warnings.append("empty_response")
    if re.search(r"\b(always|everyone must|exactly)\b", text, re.I):
        warnings.append("possible_overgeneralization")
    if re.search(r"\bdrink (?:at least )?\d+(?:,\d{3})?\s*ml\b", text, re.I) and "reference" not in text.lower():
        warnings.append("check_if_target_is_presented_as_universal")
    if "doctor" in text.lower() or "clinician" in text.lower():
        warnings.append("medical_caveat_present")
    return {"ok": not warnings, "warnings": warnings}


def aq_debug_answer(question, ctx=None):

def _aq_semantic_intent(question):
    """Supplement classifier with phrase catalog and a specificity score."""
    q = _aq_normalize_spelling(question)
    best = (0, "general_or_unknown")
    label_map = {
        "daily_amount": "daily_water_amount",
        "age_questions": "age_based_intake",
        "during_activity": "exercise_hydration",
        "before_activity": "exercise_hydration",
        "after_activity": "exercise_hydration",
        "hot_humid": "weather",
        "symptoms": "dehydration",
        "haze": "haze_hydration",
        "overhydration": "medical_safety",
        "bottle_and_tracking": "hydration_tracking",
        "diet_and_drinks": "beverages",
        "routine": "drinking_schedule",
    }
    for group, phrases in AQ_SEMANTIC_PHRASES.items():
        for phrase in phrases:
            if _aq_has_phrase(q, phrase):
                score = len(_aq_normalize(phrase))
                if score > best[0]:
                    best = (score, label_map[group])
    return best[1]


# Keep the primary classifier simple and deterministic, but let the expanded
# catalogue recover phrasings that were not caught by the original alias groups.
_AQ_BASE_CLASSIFY_INTENT = aq_classify_intent


def aq_classify_intent(question):
    base_label = _AQ_BASE_CLASSIFY_INTENT(question)
    if base_label != "general_or_unknown":
        return base_label
    return _aq_semantic_intent(question)


# Integrate the extended FAQ as a final offline knowledge layer. Specific intents
# still run first; generic keyword fallback remains last for compatibility.
_AQ_ROUTED_COACH_REPLY = coach_reply


def coach_reply(question, ctx):
    context = aq_validate_context(ctx)
    routed = _aq_route_question(question, context)
    if routed:
        if _aq_language(question) == "ms" and routed[:1].isascii():
            routed = _aq_translate_core_to_malay(routed, question)
        return routed
    extended = _aq_extended_faq_answer(question, context)
    if extended:
        return extended
    # Common informal typo phrases are tried one final time against the semantic
    # catalogue. This does not rewrite numeric details or override safety rules.
    semantic = _aq_semantic_intent(question)
    if semantic == "age_based_intake":
        return _aq_age_answer(question, context) or _aq_daily_intake_answer(question, context)
    if semantic == "exercise_hydration":
        return _aq_exercise_answer(question, context)
    if semantic == "drinking_schedule":
        return _aq_schedule_answer(question, context)
    return _AQ_ROUTED_COACH_REPLY(question, context)

def aq_run_haze_regression_matrix():
    """Run all haze variants offline and report intent/content failures."""
    context = {"total": 0, "goal": 2000, "cup": 200, "condition": "Default"}
    rows = []
    for question in AQ_HAZE_REGRESSION_QUESTIONS:
        intent = aq_classify_intent(question)
        answer = coach_reply(question, context)
        lower = str(answer or '').lower()
        rows.append({
            'question': question,
            'intent': intent,
            'intent_pass': intent == 'haze_hydration',
            'answer_nonempty': bool(answer and answer.strip()),
            'mentions_haze_or_air_quality': any(term in lower for term in ('haze', 'smog', 'air quality', 'jerebu', 'udara')),
            'avoids_false_protection_claim': not any(term in lower for term in ('flushes haze from your lungs', 'prevents pm2.5', 'removes pm2.5 from your lungs')),
        })
    return rows
